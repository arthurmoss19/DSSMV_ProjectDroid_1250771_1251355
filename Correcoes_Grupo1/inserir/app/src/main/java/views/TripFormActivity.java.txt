    /** Abre a viagem que veio no Intent (modo de edição). */
    private void loadTrip(String tripId) {
        toolbar.setTitle(R.string.trips_edit_title);
        showLoading(true);

        tripController.getTrip(tripId, new ApiCallback<Trip>() {
            @Override
            public void onSuccess(Trip trip) {
                showLoading(false);
                if (trip == null) {
                    UiUtils.toast(TripFormActivity.this, getString(R.string.trips_load_error));
                    finish();
                    return;
                }
                editingTrip = trip;
                editTitle.setText(trip.getTitle());
                editDescription.setText(trip.getDescription());
                startDateIso = trip.getStartDate();
                endDateIso = trip.getEndDate();
                showDates();
            }

            @Override
            public void onError(String message) {
                showLoading(false);
                UiUtils.toast(TripFormActivity.this, message);
                finish();
            }
        });
    }

    /** As datas não são escritas à mão: abrem um calendário. */
    private void pickDate(boolean isStart) {
        String current = isStart ? startDateIso : endDateIso;
        Calendar calendar = Calendar.getInstance();
        Date parsed = DateUtils.parseDate(current);
        if (parsed != null) {
            calendar.setTime(parsed);
        }

        DatePickerDialog dialog = new DatePickerDialog(this,
                (view, year, month, dayOfMonth) -> {
                    String iso = String.format(Locale.US, "%04d-%02d-%02d",
                            year, month + 1, dayOfMonth);
                    if (isStart) {
                        startDateIso = iso;
                    } else {
                        endDateIso = iso;
                    }
                    showDates();
                },
                calendar.get(Calendar.YEAR),
                calendar.get(Calendar.MONTH),
                calendar.get(Calendar.DAY_OF_MONTH));
        dialog.setTitle(getString(R.string.trips_pick_date));
        dialog.show();
    }

    private void submit(Trip target) {
        showLoading(true);

        ApiCallback<Trip> callback = new ApiCallback<Trip>() {
            @Override
            public void onSuccess(Trip data) {
                showLoading(false);
                UiUtils.toast(TripFormActivity.this, getString(R.string.trips_saved));
                setResult(RESULT_OK);
                finish();
            }

            @Override
            public void onError(String message) {
                showLoading(false);
                UiUtils.toast(TripFormActivity.this, message);
            }
        };

        tripController.save(target, callback);
    }
